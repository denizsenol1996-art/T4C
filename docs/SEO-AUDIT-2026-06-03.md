## 2. SEO-bevindingen (gerubriceerd)

### 2.A INDEXATIE-INFRA (P0 — MOET nu)

#### 2.A.1 Sitemap = 404
**Probleem**: geen sitemap.xml endpoint of file. Cloudflare-default robots.txt heeft ook geen `Sitemap:`-directive.
**Gevolg**: Google moet voorraad ontdekken door random links te volgen. Bij 0 inbound links + querystring-only detail-URLs = praktisch onvindbaar.

**FIX** (nieuw bestand `/opt/t4c/backend/routes/seo.js`):
```js
// routes/seo.js — dynamic sitemap + robots.txt
const router = require("express").Router()
const { queryAll } = require("../db")

const BASE = process.env.PUBLIC_URL || "https://transfer4cars.com"

function slugify(s) {
  return String(s || "").toLowerCase()
    .replace(/[^a-z0-9]+/g, "-").replace(/^-|-$/g, "").slice(0, 60)
}

router.get("/sitemap.xml", (req, res) => {
  const statics = [
    { loc: "/",           prio: "1.0", chg: "daily"   },
    { loc: "/aanbod/",    prio: "0.9", chg: "hourly"  },
    { loc: "/over-ons",   prio: "0.6", chg: "monthly" },
    { loc: "/diensten",   prio: "0.7", chg: "monthly" },
    { loc: "/contact",    prio: "0.5", chg: "monthly" },
    { loc: "/transport",  prio: "0.6", chg: "monthly" },
    { loc: "/privacy",    prio: "0.3", chg: "yearly"  },
    { loc: "/voorwaarden",prio: "0.3", chg: "yearly"  },
  ]
  let cars = []
  try {
    cars = queryAll(
      "SELECT id, merk, model, bouwjaar, updated_at FROM dv_vehicles WHERE status='active' ORDER BY updated_at DESC"
    )
  } catch (e) {}

  const urls = [
    ...statics.map(s => `  <url><loc>${BASE}${s.loc}</loc><changefreq>${s.chg}</changefreq><priority>${s.prio}</priority></url>`),
    ...cars.map(c => {
      const slug = slugify(`${c.merk}-${c.model}-${c.bouwjaar}`)
      const last = (c.updated_at || "").slice(0, 10) || new Date().toISOString().slice(0,10)
      return `  <url><loc>${BASE}/auto/${slug}-${c.id}</loc><lastmod>${last}</lastmod><changefreq>weekly</changefreq><priority>0.8</priority></url>`
    })
  ].join("\n")

  res.set("Content-Type", "application/xml; charset=utf-8")
  res.set("Cache-Control", "public, max-age=3600")
  res.send(`<?xml version="1.0" encoding="UTF-8"?>\n<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">\n${urls}\n</urlset>`)
})

router.get("/robots.txt", (req, res) => {
  res.set("Content-Type", "text/plain; charset=utf-8")
  res.send([
    "User-agent: *",
    "Allow: /",
    "Disallow: /admin/",
    "Disallow: /api/",
    "Disallow: /app/",
    "Disallow: /lyra-ai/",
    "Disallow: /telex-inkoop",
    "Disallow: /verkoop/",
    "",
    "User-agent: GPTBot",
    "Disallow: /",
    "User-agent: CCBot",
    "Disallow: /",
    "User-agent: ClaudeBot",
    "Disallow: /",
    "User-agent: Google-Extended",
    "Disallow: /",
    "",
    `Sitemap: ${BASE}/sitemap.xml`,
    ""
  ].join("\n"))
})

module.exports = router
```

**Mount in `/opt/t4c/backend/server.js`** vóór de catch-all (vóór line 158):
```js
app.use(require("./routes/seo"))
```

**LET OP**: Cloudflare serveert nu de managed-default robots.txt. Na deploy moet je in Cloudflare-dashboard onder "Crawl Control / Bot Management" of "Cache Rules" een **bypass** zetten voor `/robots.txt` zodat Express' versie doorkomt. Anders blijft de managed-default winnen.

#### 2.A.2 robots.txt zonder Sitemap-directive
Cloudflare-default heeft alleen `Allow: /` en Content-Signals. Geen `Sitemap:`-regel. Google MOET de sitemap-URL kunnen vinden.
**Fix**: zit in 2.A.1.

#### 2.A.3 /admin/ publiek 200 ⚠️ KRITIEK
**Probleem**: `app.use("/admin", express.static(...))` serveert `cardatax/admin/index.html` (68 KB, titel "T4C Command Center") aan iedereen. Auth zit pas in de JS (client-side check), niet voor de HTML-serve.
**Risico**:
- Google indexeert beheerpaneel-URL
- Concurrent ziet je tech-stack
- Bots/scrapers scannen het admin-skelet voor bekende kwetsbaarheden

**FIX optie A** (snel, minimaal): voeg X-Robots-Tag header toe in de static-mount.
```js
// server.js line 155 — vervang door:
app.use("/admin", express.static(path.join(__dirname, "..", "sites", "cardatax", "admin"), {
  extensions: ["html"],
  setHeaders: (res, p) => {
    res.set("X-Robots-Tag", "noindex, nofollow, noarchive")
    if (p.endsWith(".html")) res.set("Cache-Control", "no-store")
  }
}))
```
Plus in admin-HTML zelf `<meta name="robots" content="noindex,nofollow">` (defense in depth).

**FIX optie B** (beter, vereist meer werk): server-side auth-gate vóór de static-serve. Reject met 401 als geen geldige admin-JWT cookie. Pas dán `sendFile`. — Doe dit na P0-stack live is.

#### 2.A.4 Wildcard fallback genereert duplicate content
**Probleem** (server.js:158-165): elke verzonnen URL (`/asdf`, `/occasions`, `/verkopen`, `/b2b`) → `sendFile(index.html)` met HTTP 200. Google ziet dezelfde homepage HTML op duizenden mogelijke URLs.

**FIX**: vervang fallback door whitelist + 404 voor de rest.
```js
// server.js — vervang lines 158-165:
const T4C_PUBLIC_PATHS = new Set([
  "/", "/over-ons", "/diensten", "/contact", "/transport",
  "/privacy", "/voorwaarden", "/verkopen", "/aanbod", "/auto"
])
app.use((req, res, next) => {
  if (req.site !== "transfer4cars") return next()
  if (req.path.startsWith("/api/") || req.path.startsWith("/photos") || req.path.startsWith("/admin")
      || req.path.startsWith("/app") || req.path.startsWith("/telex-inkoop")
      || /^\/(inbound|inbound-detail|inbound-render)\.(js|css)$/.test(req.path)
      || req.path === "/admin.css") return next()
  // Try static first (handles /aanbod/, /auto/, /veilingen/, assets)
  express.static(T4C_SALES_DIR, { extensions: ["html"] })(req, res, () => {
    // Whitelist-only fallback to homepage
    const top = "/" + (req.path.split("/")[1] || "")
    if (T4C_PUBLIC_PATHS.has(top) && !req.path.includes(".")) {
      return res.sendFile(path.join(T4C_SALES_DIR, "index.html"))
    }
    // Everything else → proper 404
    res.status(404).sendFile(path.join(T4C_SALES_DIR, "404.html"), (err) => {
      if (err) res.status(404).send("Pagina niet gevonden")
    })
  })
})
```
Plus bouw `/opt/t4c/sites/transfer4cars/404.html` (kan kopie van homepage-shell zijn met `<meta name="robots" content="noindex">` + duidelijke "404" h1).

#### 2.A.5 /lyra-ai/ valt op pad-miss terug op homepage
**Probleem**: `/lyra-ai/` zonder verdere path returnt T4C homepage (omdat lyraProxy faalt en geen catch → static-fallback grijpt). Cosmetisch raar, en Google kan deze URL indexeren met homepage-content.
**Fix**: blok in robots.txt (zit in 2.A.1), plus optioneel X-Robots-Tag header in lyra-proxy middleware.

### 2.B ON-PAGE SEO (P0 + P1)

#### 2.B.1 Auto-detail = querystring → 1 URL voor 40 auto's ⚠️ ZEER GROOT
**Status nu**:
- Lijst: `/aanbod/index.html` fetcht `/api/dv/voorraad`, rendert client-side cards die linken naar `/aanbod/?id=114`
- Detail: `/aanbod/?id=114` is **dezelfde HTML** (geen redirect). JS leest `?id=` en fetcht `/api/dv/voorraad/114` of vergelijkbaar.
- Apart `/auto/?id=114` bestaat als zelfstandige detail-page met `/api/public/voorraad/:id`.

**Probleem**: Google indexeert query-string varianten zelden goed; bij ontbrekende canonical = grote kans op 1 indexed URL voor alles. Long-tail "volvo v50 2.4 edition I automaat ter aar" wordt nooit getoond.

**FIX (P0)** — twee acties:

**A. Server-side rewrite** `/auto/{slug}-{id}` → serve `/opt/t4c/sites/transfer4cars/auto/index.html` met SSR-meta-injectie.

Voeg toe in `/opt/t4c/backend/routes/seo.js`:

let _autoTmpl = null
function autoTmpl() { return _autoTmpl ||= fs.readFileSync(AUTO_HTML, "utf8") }

router.get("/auto/:slug", (req, res, next) => {
  // Match "merk-model-jaar-ID" at end → numeric ID
  const m = req.params.slug.match(/-(\d+)$/)

  let car

  } catch (e) {}
  if (!car) return res.status(404).send("Voertuig niet gevonden")

  // Bouw SEO-meta
  const title = `${car.merk} ${car.model} ${car.bouwjaar} — €${Number(car.prijs||0).toLocaleString("nl-NL")} | Transfer4Cars`
  const desc  = `${car.merk} ${car.model} ${car.type||""}, bouwjaar ${car.bouwjaar}, ${(car.km||0).toLocaleString("nl-NL")} km, ${car.brandstof||""}. Direct beschikbaar bij Transfer4Cars, Ter Aar.`.replace(/\s+/g, " ").trim()
  let cover = ""
  try {
    const fotos = JSON.parse(car.afbeeldingen || "[]")
    cover = fotos[0]?.url || ""
  } catch(e) {}
  const canonical = `https://transfer4cars.com/auto/${req.params.slug}`

  const jsonLd = {
    "@context": "https://schema.org",
    "@type": "Vehicle",
    "name": `${car.merk} ${car.model} ${car.bouwjaar}`,
    "brand": { "@type": "Brand", "name": car.merk },
    "model": car.model,
    "vehicleModelDate": car.bouwjaar,

      "@type": "Offer",
      "price": car.prijs || car.meeneemprijs || 0,
      "priceCurrency": "EUR",
      "availability": "https://schema.org/InStock",
      "seller": { "@type": "AutoDealer", "name": "Transfer4Cars" }

  const head = `
<title>${title}</title>
<meta name="description" content="${desc.replace(/"/g,'&quot;')}">
<link rel="canonical" href="${canonical}">
<meta property="og:type" content="product">
<meta property="og:title" content="${title.replace(/"/g,'&quot;')}">
<meta property="og:description" content="${desc.replace(/"/g,'&quot;')}">

<meta property="og:locale" content="nl_NL">

  // Vervang de bestaande <title>...</title> in template door onze head-blok
  const html = autoTmpl().replace(/<title>.*?<\/title>\s*<meta name="description"[^>]*>/s, head)
  res.set("Cache-Control", "public, max-age=600, s-maxage=3600")
